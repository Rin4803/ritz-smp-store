# Coins System - RitzSMP
# ย่อจำนวนเหรียญให้เป็น K / M / B / T ใช้ %js_coins% ในสกอร์บอร์ด

# Function ย่อตัวเลข (รองรับ K, M, B, T)
function fmt_num(n: number) :: text:
    if {_n} >= 1000000000000:
        set {_f} to round({_n} / 1000000000000 * 10) / 10
        return "%{_f}%T"
    else if {_n} >= 1000000000:
        set {_f} to round({_n} / 1000000000 * 10) / 10
        return "%{_f}%B"
    else if {_n} >= 1000000:
        set {_f} to round({_n} / 1000000 * 10) / 10
        return "%{_f}%M"
    else if {_n} >= 1000:
        set {_f} to round({_n} / 1000 * 10) / 10
        return "%{_f}%K"
    else:
        return "%{_n}%"

# %js_coins% - ย่อแล้ว
placeholderapi placeholder with the prefix "js":
    if the identifier is "coins":
        set {_s} to placeholder "playerpoints_points" from player
        if {_s} is not set:
            set the result to "0"
            stop
        set {_n} to {_s} parsed as number
        if {_n} is not set:
            set the result to "0"
            stop
        set the result to fmt_num({_n})

# %js_coins_raw% - เลขจริงๆ
placeholderapi placeholder with the prefix "js":
    if the identifier is "coins_raw":
        set {_s} to placeholder "playerpoints_points" from player
        if {_s} is not set:
            set the result to "0"
            stop
        set {_n} to {_s} parsed as number
        if {_n} is not set:
            set the result to "0"
            stop
        set the result to "%{_n}%"
===== /plugins/Skript/scripts/combat.sk =====
# ===== Combat Tag System - RitzSMP =====
# โดนผู้เล่นตี = ติด Combat 50 วิ
# โดนซ้ำตอนติด = รีเซ็ตเป็น 50 วิใหม่
# damage from mob / fall / lava / etc. ไม่นับ
# แสดง actionbar นับถอยหลัง
# ออกจากเกมขณะติด Combat = กลับเข้ามาตายทันที

on damage:
    attacker is a player
    victim is a player
    damage cause is entity attack
    
    set {combat::%victim's uuid%} to unix timestamp of now
    send actionbar "&c⚔ Combat &750s" to victim

every 1 second:
    loop all players:
        if {combat::%loop-player's uuid%} is set:
            set {_remain} to ({combat::%loop-player's uuid%} + 50) - unix timestamp of now
            
            if {_remain} <= 0:
                delete {combat::%loop-player's uuid%}
                send actionbar "&a✓ &7ออกจาก Combat แล้ว" to loop-player
                wait 1 second
                send actionbar "" to loop-player
            else:
                send actionbar "&c⚔ Combat &7%round({_remain})%s" to loop-player

# ===== ตาย = เคลียร์ Combat =====
on death:
    set {_vic} to victim
    if {combat::%{_vic}'s uuid%} is set:
        set {_remain} to ({combat::%{_vic}'s uuid%} + 50) - unix timestamp of now
        if {_remain} > 0:
            send actionbar "&c✖ &7Combat ถูกยกเลิก — คุณตาย!" to {_vic}
        delete {combat::%{_vic}'s uuid%}

# ===== ออกเกมขณะติด Combat =====
on quit:
    if {combat::%player's uuid%} is set:
        set {_remain} to ({combat::%player's uuid%} + 50) - unix timestamp of now
        if {_remain} > 0:
            # Mark ว่า combat-flee แล้วลบ combat tag
            set {combatflee::%player's uuid%} to true
            delete {combat::%player's uuid%}

# ===== เข้ามาใหม่ = โดน kill ทันที =====
on join:
    if {combatflee::%player's uuid%} is true:
        delete {combatflee::%player's uuid%}
        wait 5 ticks
        make console execute command "kill %player%"
        broadcast "&c&l☠ &7%player% &cหนีตายขณะติด Combat! ถูกฆ่าเรียบร้อย"

# Crate Menu RitzSMP - ใช้เหรียญ PlayerPoints
# คลิก Shulker Box ใน Lobby เท่านั้น

on script load:
    # ราคากล่อง
    set {crate_price::Common} to 50
    set {crate_price::Epic} to 100
    set {crate_price::Rare} to 200
    set {crate_price::Legend} to 400
    set {crate_price::Spawn} to 500

    # ชื่อ crate ต้องตรงกับไฟล์จริงของ CrazyCrates
    set {crate_id::Common} to "CommonBox"
    set {crate_id::Epic} to "Epic"
    set {crate_id::Rare} to "RareBox"
    set {crate_id::Legend} to "LegendaryBox"
    set {crate_id::Spawn} to "SpawnerBox"

# ===== คลิกเฉพาะ Shulker Box กล่องสุ่มใน Lobby =====
on right click:
    if player's world is "Lobby":
        set {_b} to target block of player
        set {_t} to type of {_b}

        if {_t} is white shulker box:
            set {_crate} to "Common"
        else if {_t} is light blue shulker box:
            set {_crate} to "Epic"
        else if {_t} is purple shulker box:
            set {_crate} to "Rare"
        else if {_t} is red shulker box:
            set {_crate} to "Legend"
        else if {_t} is pink shulker box:
            set {_crate} to "Spawn"
        else:
            stop

        # ยกเลิกการเปิด Shulker Box (กัน CrazyCrates รับ event)
        cancel event

        # ── Cooldown 1 วินาทีแบบ Timestamp (กันคลิกถี่) ──
        if {crate_click_cooldown::%player's uuid%} is set:
            set {_last} to {crate_click_cooldown::%player's uuid%}
            set {_diff} to difference between {_last} and now
            if {_diff} < 1 second:
                stop
        set {crate_click_cooldown::%player's uuid%} to now

        # ตรวจสอบราคา
        set {_price} to {crate_price::%{_crate}%}
        if {_price} is not set:
            send "&c❌ กล่องนี้ยังไม่มีราคา" to player
            stop

        # ดึงยอด PlayerPoints
        set {_bal} to placeholder "playerpoints_points" from player
        if {_bal} is not set:
            set {_bal} to "0"
        set {_bal} to {_bal} parsed as number
        if {_bal} is not set:
            set {_bal} to 0

        if {_bal} < {_price}:
            send "&c❌ เหรียญไม่พอ! ต้องการ &6%fmt_num({_price})% &cเหรียญ (คุณมี &e%fmt_num({_bal})%&c)" to player
            stop

        # ตรวจสอบ Crate ID
        set {_crate_id} to {crate_id::%{_crate}%}
        if {_crate_id} is not set:
            send "&c❌ ไม่พบกล่องนี้ในระบบ" to player
            stop

        # หักเหรียญก่อนเปิด
        make console execute command "points take %player's name% %{_price}%"

        # ให้ virtual key + เปิดกล่อง
        make console execute command "crazycrates give virtual %{_crate_id}% 1 %player's name%"
        wait 1 tick
        make console execute command "crazycrates forceopen %{_crate_id}% %player's name%"

        send "&a✦ ใช้ &6%fmt_num({_price})% &aเหรียญ เปิดกล่อง &e%{_crate}% &aแล้ว!" to player

# ===== Sync Balance ตอนผู้เล่นเข้า =====
on join:
    delete {bal::%player%}

# RTP v2.1 - RitzSMP (Fixed with countdown & cooldown)
options:
    rtp-world: Survival
    rtp-radius: 25000

command /rtp:
    executable by: players
    trigger:
        # ── ใช้งานได้เฉพาะโลก Survival เท่านั้น ──
        if name of player's world is not "Survival":
            send "&c✗ คำสั่ง /rtp ใช้ได้เฉพาะในโลก Survival เท่านั้น" to player
            stop

        # ── Cooldown Check ──
        if {cooldown::rtp::%player%} > now:
            set {_rem} to difference between {cooldown::rtp::%player%} and now
            send "&c⏳ กรุณารอ &6%{_rem}% &cวินาทีก่อนใช้ /rtp อีกครั้ง" to player
            stop

        # ── Combat Check ──
        if {combat::%player's uuid%} is set:
            send "&c✖ คุณอยู่ในโหมดต่อสู้! วาร์ปไม่ได้" to player
            stop

        # ── เช็คโลก Survival ──
        set {_world} to world("{@rtp-world}")
        if {_world} is not set:
            send "&c❌ ไม่พบโลก Survival กรุณาติดต่อแอดมิน" to player
            stop

        set {_negRadius} to -25000
        set {_posRadius} to 25000

        loop 10 times:
            set {_x} to random integer between {_negRadius} and {_posRadius}
            set {_z} to random integer between {_negRadius} and {_posRadius}
            set {_top} to highest solid block at location({_x}, 0, {_z}, {_world})
            set {_block} to block at location({_x}, y-coordinate of {_top}, {_z}, {_world})
            if {_block} is water or lava:
                continue

            set {_target} to location({_x} + 0.5, y-coordinate of {_top} + 1, {_z} + 0.5, {_world})

            # ── เริ่ม countdown 3 วิ ──
            set {_start} to location of player
            send "&a✦ &fกำลังวาร์ปไปที่ &eX:%{_x}% Z:%{_z}% &f... &7(ห้ามขยับ)"
            set {_cd} to 3
            loop 3 times:
                if distance between player and {_start} > 0.5:
                    send "&c✗ &fวาปยกเลิก &8— &fคุณขยับ!" to player
                    stop
                if {combat::%player's uuid%} is set:
                    send "&c✗ &fคุณติด Combat ระหว่างวาป! &8— &fยกเลิก" to player
                    stop
                send actionbar "&a⏳ &2%{_cd}% &fวิ &8| &cอย่าขยับ!" to player
                wait 1 second
                subtract 1 from {_cd}

            # ── วาร์ปสำเร็จ ──
            teleport player to {_target}
            set {_px} to x-coordinate of player
            set {_py} to y-coordinate of player + 1
            set {_pz} to z-coordinate of player
            execute console command "/particle witch %{_px}% %{_py}% %{_pz}% 1 1 1 0 15"
            play sound "entity.enderman.teleport" to player
            send "&a✦ &fสุ่มวาร์ปไปที่ &eX:%{_x}% Z:%{_z}% &fสำเร็จ!" to player

            # ── Cooldown 60 วินาที ──
            set {cooldown::rtp::%player%} to now + 60 seconds
            stop

        # ── หาพื้นที่ไม่ได้ ──
        send "&c✗ ไม่พบพื้นที่ปลอดภัยหลังจากพยายาม 10 ครั้ง" to player
